# Define source files
set(Sources
    RadixSortGPU.cpp
    ComputeDeviceData.cpp
    Dataset.cpp
    HostData.cpp
)

add_library(radixsortcl STATIC
    ${Sources}
)

target_include_directories(radixsortcl
PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:include>
)

# Link required libraries
target_link_libraries(radixsortcl
PUBLIC
    GPUCommon
# OpenCL::UtilsCpp is used directly in radixsortcl's .cpp files but must not
# appear in the installed interface (it's a Khronos SDK-specific target).
PRIVATE
    $<BUILD_INTERFACE:OpenCL::UtilsCpp>
)

set_source_files_properties("${Sources}"
    PROPERTIES
        COMPILE_FLAGS
            "$<IF:$<CXX_COMPILER_ID:MSVC>,\
            /W4;$<$<CONFIG:RELEASE>:/O2>,\
            -Wall;-Wextra;-Werror;-pedantic;>"
)

# ---------------------------------------------------------------------------
# Pre-compile RadixSort.cl for each supported data type.
#
# The compile_opencl_kernels() function (from cmake/CompileOpenCLKernel.cmake)
# will:
#   1. Detect available offline OpenCL tools (clang -target spir64 / clspv /
#      clang+llvm-spirv for SPIR-V; clang for native binary).
#   2. Emit add_custom_command rules that produce per-type kernel files
#      in ${CMAKE_CURRENT_BINARY_DIR}/kernels/.
#   3. Create a radixsortcl_kernels aggregate target that radixsortcl depends on.
#   4. Set RADIXSORTCL_KERNEL_DIR / RADIXSORTCL_KERNEL_EXT compile definitions
#      so RadixSortGPU.cpp knows where to look at runtime.
#
# If no offline compiler is found the function emits a warning and returns
# without adding any rules; the library will fall back to runtime source
# compilation as before.
# ---------------------------------------------------------------------------
include(CompileOpenCLKernel)

compile_opencl_kernels(
    TARGET        radixsortcl
    KERNEL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/kernels/RadixSort.cl"
    OUTPUT_DIR    "${CMAKE_CURRENT_BINARY_DIR}/kernels"
    DATA_TYPES    int32 uint32 int64 uint64
)

# ---------------------------------------------------------------------------
# Install rules and package export
# ---------------------------------------------------------------------------

set(_cmake_install_dir "lib/cmake/radixsortcl")

# Static library with export set
install(
    TARGETS radixsortcl
    EXPORT  radixsortclTargets
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin
)

include(CMakePackageConfigHelpers)

# --- Installed-tree package config ---
configure_package_config_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/radixsortclConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/radixsortclConfig.cmake"
    INSTALL_DESTINATION "${_cmake_install_dir}"
)

write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/radixsortclConfigVersion.cmake"
    VERSION       ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)

install(
    FILES
        "${CMAKE_CURRENT_BINARY_DIR}/radixsortclConfig.cmake"
        "${CMAKE_CURRENT_BINARY_DIR}/radixsortclConfigVersion.cmake"
    DESTINATION "${_cmake_install_dir}"
)

install(
    EXPORT      radixsortclTargets
    NAMESPACE   radixsortcl::
    DESTINATION "${_cmake_install_dir}"
)

# CompileOpenCLKernel module — installed alongside the config file so that
# radixsortclConfig.cmake can append its install dir to CMAKE_MODULE_PATH.
install(
    FILES "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/CompileOpenCLKernel.cmake"
    DESTINATION "${_cmake_install_dir}"
)

# --- Build-tree package config (enables find_package from the build tree) ---
# Used by CPM / FetchContent via CMAKE_FIND_PACKAGE_REDIRECTS_DIR (CMake 3.24+)
# and by callers that set radixsortcl_DIR to the build directory manually.
#
# Written to a subdirectory so it doesn't overwrite the installed-tree config
# that configure_package_config_file() placed in CMAKE_CURRENT_BINARY_DIR.
# Set radixsortcl_DIR to ${CMAKE_CURRENT_BINARY_DIR}/package to consume it.
export(
    EXPORT    radixsortclTargets
    NAMESPACE radixsortcl::
    FILE      "${CMAKE_CURRENT_BINARY_DIR}/package/radixsortclTargets.cmake"
)

# The build-tree config adds the source cmake/ directory to CMAKE_MODULE_PATH
# so consumers get CompileOpenCLKernel without any install step.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package/radixsortclConfig.cmake"
    "# Build-tree config — generated by CMake, do not edit.\n"
    "include(CMakeFindDependencyMacro)\n"
    "list(APPEND CMAKE_MODULE_PATH \"${CMAKE_CURRENT_SOURCE_DIR}/../cmake\")\n"
    "include(\"\${CMAKE_CURRENT_LIST_DIR}/radixsortclTargets.cmake\")\n"
    "check_required_components(radixsortcl)\n"
)

# OpenCL source kernel (runtime JIT fallback)
install(
    FILES kernels/RadixSort.cl
    DESTINATION share/radixsortcl/kernels
)

# Pre-compiled kernel binaries (when offline compilation succeeded)
if(DEFINED RADIXSORTCL_COMPILED_KERNELS AND RADIXSORTCL_COMPILED_KERNELS)
    install(
        FILES ${RADIXSORTCL_COMPILED_KERNELS}
        DESTINATION share/radixsortcl/kernels
    )
endif()
